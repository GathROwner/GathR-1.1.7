it('observes a stale refresh hiding retained clusters when the query cache is absent',async()=>{
 jest.useFakeTimers();jest.setSystemTime(new Date(2026,10,14,12));
 const oldClient=(global as any).__RQ_CLIENT;
 const v={id:'retained',type:'event',title:'Retained fixture',description:'',category:'Live Music',venue:'Fixture venue',address:'Test',latitude:46.23,longitude:-63.12,startDate:'2026-11-14',endDate:'2026-11-14',startTime:'19:00',endTime:'23:00'} as Event;
 const event={...v,timing:createLegacyTimingContract(v)};
 const cluster={id:'retained-cluster',clusterType:'single',venues:[{locationKey:'retained',venue:'Fixture venue',events:[event]}]} as any;
 let finish:(r:any)=>void=()=>{};
 const fetchQuery=jest.fn(()=>new Promise(resolve=>{finish=resolve}));
 (global as any).__RQ_CLIENT={getQueryData:()=>undefined,fetchQuery};
 useMapStore.setState({allEvents:[event],events:[event],viewportEvents:[event],onScreenEvents:[event],filteredEvents:[event],clusters:[cluster],isLoading:false,lastFetchedAt:Date.now()-4*60*1000});
 try{
  const pending=useMapStore.getState().prefetchIfStale(180000);
  const state=useMapStore.getState();
  const duringRefresh={cachedEvents:state.allEvents.length,viewportEvents:state.viewportEvents.length,clusters:state.clusters.length,isLoading:state.isLoading,mapMarkerRenderGate:!state.isLoading};
  expect(fetchQuery).toHaveBeenCalledTimes(1);
  expect(duringRefresh).toEqual({cachedEvents:1,viewportEvents:1,clusters:1,isLoading:true,mapMarkerRenderGate:false});
  finish({combinedData:[event],fetchedAt:Date.now()});await pending;
  expect(useMapStore.getState().isLoading).toBe(false);
  require('fs').writeFileSync(require('path').join(__dirname,'../retained-cluster-refresh.json'),JSON.stringify({method:'Actual store with retained events/clusters, absent React Query entry, old lastFetchedAt and controlled pending refresh; no real network or device timings',duringRefresh,afterRefresh:{isLoading:useMapStore.getState().isLoading,clusters:useMapStore.getState().clusters.length},sourceLimits:'Root foreground fetchQuery callback alone does not use this loading path; tab post-switch prefetchIfStale or a store fetchEvents caller is required.'},null,2));
 }finally{(global as any).__RQ_CLIENT=oldClient;jest.useRealTimers();}
});
